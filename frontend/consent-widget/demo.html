Installation

Install via NPM:

npm install @datasheild/consent-widget

Usage - Script Tag

<script src="https://cdn.datasheild.in/consent-widget.min.js"></script>
<script>
  window.DataShieldConfig = {
    apiUrl: 'https://api.datasheild.in/v1',
    tenantId: 'your-tenant-id',
    purposes: [
      {
        id: 'essential',
        name: 'Essential Services',
        description: 'Required for platform functionality',
        required: true,
        category: 'ESSENTIAL'
      },
      {
        id: 'marketing',
        name: 'Marketing Communications',
        description: 'Send promotional emails and offers',
        required: false,
        category: 'MARKETING'
      }
    ],
    theme: 'light',
    position: 'bottom',
    onConsentGranted: (consent) => {
      console.log('Consent granted:', consent);
    }
  };
</script>

Usage - Module Import

import { DataShieldConsentWidget } from '@datasheild/consent-widget';

const widget = new DataShieldConsentWidget({
  apiUrl: 'https://api.datasheild.in/v1',
  tenantId: 'your-tenant-id',
  purposes: [
    {
      id: 'analytics',
      name: 'Analytics',
      description: 'Help us improve our services',
      required: false,
      category: 'ANALYTICS'
    }
  ],
  onConsentGranted: (consent) => {
    // Handle consent
    console.log('User granted consent:', consent);
  },
  onConsentWithdrawn: (consentId) => {
    // Handle withdrawal
    console.log('User withdrew consent:', consentId);
  }
});

widget.init();
